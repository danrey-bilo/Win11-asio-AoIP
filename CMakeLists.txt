cmake_minimum_required(VERSION 3.20)
project(Win11AsioAoIP VERSION 2.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 (AMD64) and a 64-bit toolchain")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]" OR CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
  message(FATAL_ERROR "Windows ARM64 is not supported")
endif()
enable_language(RC)
include(cmake/AoIPDependency.cmake)
set(ASIO_SDK_DIR "$ENV{ASIO_SDK_DIR}" CACHE PATH "External Steinberg ASIO SDK root; not distributed here")
if(NOT EXISTS "${ASIO_SDK_DIR}/common/iasiodrv.h" OR NOT EXISTS "${ASIO_SDK_DIR}/common/asio.h")
  message(FATAL_ERROR "Set ASIO_SDK_DIR to a separately obtained Steinberg ASIO SDK. See docs/ASIO-SDK.md")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(PiAoipAsio SHARED src/driver/asio_driver.cpp src/config/settings.cpp src/control/discovery.cpp src/ui/settings_panel.cpp src/platform/realtime.cpp src/ui/controls.rc src/driver/PiAoipAsio.def)
target_compile_features(PiAoipAsio PRIVATE cxx_std_17)
target_include_directories(PiAoipAsio PRIVATE "${ASIO_SDK_DIR}/common" src/ui)
target_link_libraries(PiAoipAsio PRIVATE AoIP::core ws2_32 ole32 advapi32 gdi32 iphlpapi uuid avrt comctl32 uxtheme shell32)
set_target_properties(PiAoipAsio PROPERTIES PREFIX "")
add_executable(PiAoipControl WIN32 apps/control_panel.cpp)
target_include_directories(PiAoipControl PRIVATE "${ASIO_SDK_DIR}/common")
target_link_libraries(PiAoipControl PRIVATE ole32 uuid)
foreach(target PiAoipAsio PiAoipControl)
  target_compile_features(${target} PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(${target} PRIVATE -static)
  endif()
endforeach()
